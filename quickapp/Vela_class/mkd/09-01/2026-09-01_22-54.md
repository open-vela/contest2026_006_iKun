## 四、Storage异常分析

### 当前已有的容错机制

```javascript
// storage.js 初始化时的容错处理
storage.get({
  key: STORAGE_KEYS.COURSES,
  success: function (data) {
    try {
      courseCache = data ? JSON.parse(data) : [...courses]  // 空数据使用默认值
    } catch (e) {
      console.error('Failed to parse courses:', e)
      courseCache = [...courses]  // 解析失败使用默认值
    }
    checkComplete()
  },
  fail: function (data, code) {
    console.error('Failed to get courses:', code)
    courseCache = [...courses]  // 读取失败使用默认值
    checkComplete()
  }
})
```

### 容错点

| 场景 | 处理方式 |
|------|----------|
| Storage 为空 | 使用默认 Mock 数据 |
| JSON.parse 失败 | 使用默认 Mock 数据 |
| storage.get 失败 | 使用默认 Mock 数据 |

## 七、考试容错分析

### getExamStatus() 容错

```javascript
export function getExamStatus(exam, now) {
  const examDate = parseDateTime(`${exam.date}T${exam.startTime}:00`)
  const examEnd = parseDateTime(`${exam.date}T${exam.endTime}:00`)

  if (now >= examEnd) return 'finished'
  if (now >= examDate) return 'ongoing'

  const examDay = new Date(exam.date)
  if (examDay.toDateString() === now.toDateString()) return 'today'

  return 'upcoming'
}
```
